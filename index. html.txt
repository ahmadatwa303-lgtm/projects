  <!-- Engine Script -->
  <script>
    // State
    let currentWorkspace = 'study'; // 'study', 'work', 'code', 'archive'
    let currentCategory = 'all';
    let searchQuery = '';
    let db = null;
    let categories = ["Computer Science", "Algorithms", "Databases", "Web Development", "Operating Systems"];
    let filesCache = [];

    // Initialize IndexedDB (High Capacity storage for real files)
    function initDB() {
      return new Promise((resolve) => {
        const req = indexedDB.open('EngineersVaultDB', 1);
        req.onupgradeneeded = (e) => {
          const database = e.target.result;
          if (!database.objectStoreNames.contains('files')) {
            database.createObjectStore('files', { keyPath: 'id' });
          }
        };
        req.onsuccess = (e) => {
          db = e.target.result;
          resolve();
        };
      });
    }

    async function loadVault() {
      // Load categories
      const savedCats = localStorage.getItem('engineers_categories');
      if (savedCats) {
        try { categories = JSON.parse(savedCats); } catch(e){}
      }

      // Load files from IndexedDB
      const transaction = db.transaction(['files'], 'readonly');
      const store = transaction.objectStore('files');
      const request = store.getAll();

      request.onsuccess = () => {
        filesCache = request.result || [];
        renderCategories();
        renderFiles();
      };
    }

    function saveCategories() {
      localStorage.setItem('engineers_categories', JSON.stringify(categories));
      renderCategories();
    }

    function setWorkspace(ws) {
      currentWorkspace = ws;
      currentCategory = 'all';

      document.querySelectorAll('.ws-btn').forEach(btn => {
        btn.className = 'ws-btn flex items-center space-x-2 px-3 py-2 rounded-lg text-xs font-medium text-slate-400 hover:bg-slate-800/60 hover:text-white transition';
      });

      const activeBtn = document.getElementById(`ws-${ws}`);
      if (activeBtn) {
        activeBtn.className = 'ws-btn flex items-center space-x-2 px-3 py-2 rounded-lg text-xs font-semibold bg-sky-950/70 text-sky-300 border border-sky-500/40';
      }

      document.getElementById('activeWorkspaceBadge').textContent = ws.toUpperCase();
      document.getElementById('activeFilterBadge').textContent = 'All Topics';
      renderFiles();
    }

    function renderCategories() {
      const container = document.getElementById('categoryNavList');
      const manualSelect = document.getElementById('manualCategory');
      container.innerHTML = '';
      manualSelect.innerHTML = '';

      // "All" item
      const allBtn = document.createElement('div');
      allBtn.className = `flex items-center justify-between rounded-lg px-3 py-1.5 text-xs font-medium cursor-pointer transition ${currentCategory === 'all' ? 'bg-sky-950/40 text-sky-400 border border-sky-500/30' : 'text-slate-400 hover:bg-slate-800/60 hover:text-slate-200'}`;
      allBtn.innerHTML = `<span>All Subjects</span>`;
      allBtn.onclick = () => { currentCategory = 'all'; document.getElementById('activeFilterBadge').textContent = 'All Topics'; renderFiles(); renderCategories(); };
      container.appendChild(allBtn);

      categories.forEach(cat => {
        const item = document.createElement('div');
        const isActive = currentCategory === cat;
        item.className = `group flex items-center justify-between rounded-lg px-3 py-1.5 text-xs font-medium cursor-pointer transition ${isActive ? 'bg-sky-950/40 text-sky-400 border border-sky-500/30' : 'text-slate-400 hover:bg-slate-800/60 hover:text-slate-200'}`;
        item.innerHTML = `
          <span class="truncate" onclick="selectCategory('${escapeHtml(cat)}')">${escapeHtml(cat)}</span>
          <button onclick="deleteCategory('${escapeHtml(cat)}', event)" class="opacity-0 group-hover:opacity-100 hover:text-red-400 transition">
            <i data-lucide="trash-2" class="w-3 h-3"></i>
          </button>
        `;
        container.appendChild(item);

        const opt = document.createElement('option');
        opt.value = cat;
        opt.textContent = cat;
        manualSelect.appendChild(opt);
      });
      lucide.createIcons();
    }

    function selectCategory(cat) {
      currentCategory = cat;
      document.getElementById('activeFilterBadge').textContent = cat;
      renderCategories();
      renderFiles();
    }

    function renderFiles() {
      const grid = document.getElementById('filesGrid');
      const emptyState = document.getElementById('emptyState');
      grid.innerHTML = '';

      let list = filesCache.filter(f => f.workspace === currentWorkspace);

      if (currentCategory !== 'all') {
        list = list.filter(f => f.category === currentCategory);
      }

      if (searchQuery) {
        list = list.filter(f => f.title.toLowerCase().includes(searchQuery) || (f.tags && f.tags.some(t => t.toLowerCase().includes(searchQuery))));
      }

      document.getElementById('resultsCount').textContent = `${list.length} items in view`;
      document.getElementById('storageSize').textContent = `${filesCache.length} Files Saved`;

      if (list.length === 0) {
        emptyState.classList.remove('hidden');
        emptyState.classList.add('flex');
      } else {
        emptyState.classList.add('hidden');
        emptyState.classList.remove('flex');

        list.forEach(file => {
          const card = document.createElement('div');
          card.className = "bg-[#101726] border border-engineers-border rounded-xl p-4 flex flex-col justify-between hover:border-sky-500/50 transition cursor-pointer group";
          card.onclick = (e) => {
            if (e.target.closest('button')) return;
            openViewer(file);
          };

          const icon = file.isCode ? 'code-2' : (file.type && file.type.includes('image') ? 'image' : 'file-text');

          card.innerHTML = `
            <div>
              <div class="flex items-center justify-between mb-2">
                <span class="text-[10px] font-mono px-2 py-0.5 rounded bg-sky-950 text-sky-400 font-semibold uppercase tracking-wider">${escapeHtml(file.category)}</span>
                <button onclick="deleteFile('${file.id}', event)" class="p-1 text-slate-500 hover:text-red-400 transition" title="Delete">
                  <i data-lucide="trash" class="w-3.5 h-3.5"></i>
                </button>
              </div>
              <div class="flex items-start space-x-2.5 mb-2">
                <i data-lucide="${icon}" class="w-5 h-5 text-sky-400 shrink-0 mt-0.5"></i>
                <h4 class="font-bold text-white text-sm tracking-tight truncate group-hover:text-sky-400 transition">${escapeHtml(file.title)}</h4>
              </div>
            </div>
            <div class="pt-3 border-t border-engineers-border flex items-center justify-between text-[11px] font-mono text-slate-500 mt-2">
              <span>${file.size || 'Text'}</span>
              <span>${new Date(file.updatedAt).toLocaleDateString()}</span>
            </div>
          `;
          grid.appendChild(card);
        });
      }
      lucide.createIcons();
    }

    // Process & Store Any File
    async function storeFile(fileObj) {
      const tx = db.transaction(['files'], 'readwrite');
      tx.objectStore('files').put(fileObj);
      tx.oncomplete = () => {
        filesCache.unshift(fileObj);
        renderFiles();
        showToast(`Stored: ${fileObj.title}`);
      };
    }

    async function handleFileInput(e) {
      const files = e.target.files;
      if (!files || files.length === 0) return;

      for (let f of files) {
        await processAndSaveFile(f);
      }
      e.target.value = '';
    }

    function processAndSaveFile(file) {
      return new Promise((resolve) => {
        const ext = file.name.split('.').pop().toLowerCase();
        const isCode = ['cpp', 'c', 'py', 'js', 'sh', 'sql', 'html', 'css', 'java', 'json', 'txt', 'md'].includes(ext);

        const reader = new FileReader();
        if (isCode) {
          reader.onload = (e) => {
            const entry = {
              id: 'file_' + Date.now() + '_' + Math.random().toString(36).substr(2, 4),
              title: file.name,
              workspace: currentWorkspace,
              category: currentCategory !== 'all' ? currentCategory : (categories[0] || 'General'),
              content: e.target.result,
              isCode: true,
              syntax: ext === 'cpp' ? 'cpp' : (ext === 'py' ? 'python' : 'markdown'),
              size: (file.size / 1024).toFixed(1) + ' KB',
              updatedAt: new Date().toISOString()
            };
            storeFile(entry);
            resolve();
          };
          reader.readAsText(file);
        } else {
          reader.onload = (e) => {
            const entry = {
              id: 'file_' + Date.now() + '_' + Math.random().toString(36).substr(2, 4),
              title: file.name,
              workspace: currentWorkspace,
              category: currentCategory !== 'all' ? currentCategory : (categories[0] || 'General'),
              content: e.target.result,
              isCode: false,
              type: file.type,
              size: (file.size / 1024).toFixed(1) + ' KB',
              updatedAt: new Date().toISOString()
            };
            storeFile(entry);
            resolve();
          };
          reader.readAsDataURL(file);
        }
      });
    }

    function saveManualNote(e) {
      e.preventDefault();
      const title = document.getElementById('manualTitle').value.trim();
      const category = document.getElementById('manualCategory').value;
      const syntax = document.getElementById('manualSyntax').value;
      const tags = document.getElementById('manualTags').value.split(',').map(t => t.trim()).filter(Boolean);
      const content = document.getElementById('manualContent').value;

      const entry = {
        id: 'doc_' + Date.now(),
        title,
        workspace: currentWorkspace,
        category,
        content,
        syntax,
        tags,
        isCode: syntax !== 'markdown',
        size: 'Doc',
        updatedAt: new Date().toISOString()
      };

      storeFile(entry);
      closeNoteModal();
    }

    function openViewer(file) {
      document.getElementById('viewTitle').textContent = file.title;
      document.getElementById('viewCategoryBadge').textContent = file.category;
      document.getElementById('viewTypeBadge').textContent = file.syntax || file.size;

      const dlBtn = document.getElementById('downloadFileBtn');
      const body = document.getElementById('viewBody');
      body.innerHTML = '';

      if (file.isCode || file.syntax) {
        body.innerHTML = `<pre><code class="language-${file.syntax || 'bash'}">${escapeHtml(file.content)}</code></pre>`;
        hljs.highlightElement(body.querySelector('code'));

        const blob = new Blob([file.content], { type: 'text/plain' });
        dlBtn.href = URL.createObjectURL(blob);
        dlBtn.download = file.title;
      } else if (file.type && file.type.startsWith('image/')) {
        body.innerHTML = `<img src="${file.content}" class="max-h-[60vh] mx-auto rounded-lg shadow-md" alt="${escapeHtml(file.title)}" />`;
        dlBtn.href = file.content;
        dlBtn.download = file.title;
      } else {
        body.innerHTML = `<div class="p-8 text-center text-slate-400"><i data-lucide="file-check" class="w-12 h-12 mx-auto mb-3 text-sky-400"></i><p>Binary or Document file safely stored in Engineers Vault.</p></div>`;
        dlBtn.href = file.content;
        dlBtn.download = file.title;
      }

      document.getElementById('viewModal').classList.remove('hidden');
      lucide.createIcons();
    }

    function deleteFile(id, e) {
      e.stopPropagation();
      if (!confirm("Delete this item permanently?")) return;
      const tx = db.transaction(['files'], 'readwrite');
      tx.objectStore('files').delete(id);
      tx.oncomplete = () => {
        filesCache = filesCache.filter(f => f.id !== id);
        renderFiles();
        showToast("File deleted");
      };
    }

    // Drag and Drop
    function setupDragAndDrop() {
      const dropZone = document.getElementById('dropZone');
      ['dragenter', 'dragover'].forEach(name => {
        window.addEventListener(name, (e) => { e.preventDefault(); dropZone.classList.add('drag-hover'); });
      });
      ['dragleave', 'drop'].forEach(name => {
        window.addEventListener(name, (e) => { e.preventDefault(); dropZone.classList.remove('drag-hover'); });
      });
      window.addEventListener('drop', async (e) => {
        e.preventDefault();
        const files = e.dataTransfer.files;
        if (files) {
          for (let f of files) await processAndSaveFile(f);
        }
      });
    }

    function exportVault() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify({ categories, files: filesCache }, null, 2));
      const dl = document.createElement('a');
      dl.setAttribute("href", dataStr);
      dl.setAttribute("download", `ENGINEERS_BACKUP_${new Date().toISOString().slice(0,10)}.json`);
      document.body.appendChild(dl);
      dl.click();
      dl.remove();
      showToast("Backup downloaded");
    }

    function showToast(msg) {
      const toast = document.getElementById('toast');
      document.getElementById('toastMsg').textContent = msg;
      toast.classList.remove('translate-y-20', 'opacity-0');
      toast.classList.add('translate-y-0', 'opacity-100');
      setTimeout(() => {
        toast.classList.add('translate-y-20', 'opacity-0');
        toast.classList.remove('translate-y-0', 'opacity-100');
      }, 2500);
    }

    function escapeHtml(str) {
      if (!str) return '';
      return String(str).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
    }
